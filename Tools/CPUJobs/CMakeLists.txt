cmake_minimum_required(VERSION 3.27)

# CPU-only baseline: no Vulkan, Slang, SDL, scene assets or optional GPU SDKs.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(MetallicCPUJobs LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 23)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    get_filename_component(METALLIC_REPOSITORY_ROOT "../.." ABSOLUTE)
    set(METALLIC_STDEXEC_ROOT "${METALLIC_REPOSITORY_ROOT}/External/stdexec" CACHE PATH "Locked stdexec checkout")
    find_package(Threads REQUIRED)
    add_library(MetallicRuntimeTask STATIC "${METALLIC_REPOSITORY_ROOT}/Source/Runtime/Task/TaskSystem.cpp")
    target_include_directories(MetallicRuntimeTask PUBLIC "${METALLIC_REPOSITORY_ROOT}/Source")
    target_include_directories(MetallicRuntimeTask SYSTEM PUBLIC "${METALLIC_STDEXEC_ROOT}/include")
    target_link_libraries(MetallicRuntimeTask PUBLIC Threads::Threads)
    if(MSVC)
        target_compile_options(MetallicRuntimeTask PUBLIC /Zc:__cplusplus /Zc:preprocessor /Zc:externConstexpr /bigobj /EHsc)
        target_compile_definitions(MetallicRuntimeTask PRIVATE NOMINMAX)
    endif()
else()
    set(METALLIC_REPOSITORY_ROOT "${CMAKE_SOURCE_DIR}")
endif()

option(METALLIC_CPU_JOBS_RANGE "Include the unified ParallelFor candidate" ON)
option(METALLIC_CPU_JOBS_TBB "Include oneTBB comparison" OFF)
set(METALLIC_CPU_JOBS_TBB_ROOT "${METALLIC_REPOSITORY_ROOT}/External/oneTBB" CACHE PATH "Locked oneTBB checkout")
if(METALLIC_CPU_JOBS_TBB AND NOT TARGET TBB::tbb)
    set(TBB_TEST OFF CACHE BOOL "" FORCE)
    set(TBB_STRICT OFF CACHE BOOL "" FORCE)
    set(TBB_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(TBBMALLOC_BUILD OFF CACHE BOOL "" FORCE)
    set(TBBMALLOC_PROXY_BUILD OFF CACHE BOOL "" FORCE)
    set(TBB_DISABLE_HWLOC_AUTOMATIC_SEARCH ON CACHE BOOL "" FORCE)
    set(TBB_ENABLE_IPO OFF CACHE BOOL "" FORCE)
    add_subdirectory("${METALLIC_CPU_JOBS_TBB_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/oneTBB" EXCLUDE_FROM_ALL)
endif()

if(TARGET MetallicRuntimeScene)
    add_executable(MetallicMeshletSceneBench "${METALLIC_REPOSITORY_ROOT}/Tools/CPUJobs/MeshletSceneBench.cpp")
    add_executable(MetallicMeshletSceneBenchLegacy
        "${METALLIC_REPOSITORY_ROOT}/Tools/CPUJobs/MeshletSceneBench.cpp"
        "${METALLIC_REPOSITORY_ROOT}/Tools/CPUJobs/LegacyMeshletSceneAdapter.cpp")
    target_compile_definitions(MetallicMeshletSceneBenchLegacy PRIVATE METALLIC_MESHLET_BENCH_LEGACY=1)
    foreach(meshlet_bench IN ITEMS MetallicMeshletSceneBench MetallicMeshletSceneBenchLegacy)
        target_link_libraries(${meshlet_bench} PRIVATE MetallicRuntimeScene MetallicRuntimeTask)
        if(MSVC)
            target_compile_options(${meshlet_bench} PRIVATE /EHsc /utf-8)
            target_compile_definitions(${meshlet_bench} PRIVATE NOMINMAX)
        endif()
        metallic_copy_openusd_runtime(${meshlet_bench})
        metallic_copy_spdlog_runtime(${meshlet_bench})
    endforeach()
endif()

add_executable(MetallicCPUJobBench "${METALLIC_REPOSITORY_ROOT}/Tools/CPUJobBench.cpp" LegacyMeshletBuildParallel.cpp)
target_include_directories(MetallicCPUJobBench PRIVATE "${METALLIC_REPOSITORY_ROOT}/Tools/CPUJobs")
target_link_libraries(MetallicCPUJobBench PRIVATE MetallicRuntimeTask)
if(METALLIC_CPU_JOBS_RANGE)
    if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
        target_sources(MetallicRuntimeTask PRIVATE "${METALLIC_REPOSITORY_ROOT}/Source/Runtime/Task/ParallelFor.cpp")
    endif()
    target_compile_definitions(MetallicCPUJobBench PRIVATE METALLIC_BENCH_HAS_PARALLEL_FOR=1)
    # Manual identity/correlation evidence; no benchmark or default-build work.
    add_executable(MetallicRangeDiagnosticsExample EXCLUDE_FROM_ALL RangeDiagnosticsExample.cpp)
    target_link_libraries(MetallicRangeDiagnosticsExample PRIVATE MetallicRuntimeTask)
    if(MSVC)
        target_compile_options(MetallicRangeDiagnosticsExample PRIVATE /EHsc /utf-8)
        target_compile_definitions(MetallicRangeDiagnosticsExample PRIVATE NOMINMAX)
    endif()
endif()
if(METALLIC_CPU_JOBS_TBB)
    target_link_libraries(MetallicCPUJobBench PRIVATE TBB::tbb)
    target_compile_definitions(MetallicCPUJobBench PRIVATE METALLIC_BENCH_HAS_TBB=1)
    if(WIN32)
        add_custom_command(TARGET MetallicCPUJobBench POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "$<TARGET_FILE:tbb>" "$<TARGET_FILE_DIR:MetallicCPUJobBench>")
    endif()
endif()
if(MSVC)
    target_compile_options(MetallicCPUJobBench PRIVATE /EHsc /utf-8)
    target_compile_definitions(MetallicCPUJobBench PRIVATE NOMINMAX)
endif()

option(METALLIC_CPU_JOBS_TESTS "Build CPU-only task regressions" ON)
if(METALLIC_CPU_JOBS_TESTS AND CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    enable_testing()
    set(METALLIC_GOOGLETEST_ROOT "${METALLIC_REPOSITORY_ROOT}/External/googletest" CACHE PATH "Locked googletest checkout")
    set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
    set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
    add_subdirectory("${METALLIC_GOOGLETEST_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/googletest" EXCLUDE_FROM_ALL)
    add_executable(MetallicTaskTests "${METALLIC_REPOSITORY_ROOT}/tests/task/TaskGraphTests.cpp")
    if(METALLIC_CPU_JOBS_RANGE)
        target_sources(MetallicTaskTests PRIVATE "${METALLIC_REPOSITORY_ROOT}/tests/task/ParallelForTests.cpp")
    endif()
    target_link_libraries(MetallicTaskTests PRIVATE MetallicRuntimeTask GTest::gtest_main)
    if(MSVC)
        target_compile_options(MetallicTaskTests PRIVATE /EHsc /utf-8)
        target_compile_definitions(MetallicTaskTests PRIVATE NOMINMAX)
    endif()
    add_test(NAME MetallicTaskTests COMMAND MetallicTaskTests --gtest_color=no)
    set_tests_properties(MetallicTaskTests PROPERTIES TIMEOUT 120)
endif()
