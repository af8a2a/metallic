if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    cmake_minimum_required(VERSION 3.27)
    project(MetallicCPUKernelTests LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 23)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    enable_testing()
    include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/CPUKernels.cmake")
    find_package(GTest CONFIG QUIET)
    if(NOT TARGET GTest::gtest_main)
        set(METALLIC_GTEST_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../External/googletest"
            CACHE PATH "GoogleTest source for the standalone CPU kernel tests")
        set(gtest_force_shared_crt ON CACHE BOOL "Use the same MSVC CRT as the tests")
        add_subdirectory("${METALLIC_GTEST_SOURCE}" googletest EXCLUDE_FROM_ALL)
    endif()
endif()

add_executable(MetallicCPUKernelTests AABBTransformTests.cpp)
target_link_libraries(MetallicCPUKernelTests PRIVATE MetallicCPUKernels GTest::gtest_main)
if(TARGET MetallicRuntimeTask)
    target_sources(MetallicCPUKernelTests PRIVATE AABBParallelForTests.cpp)
    target_link_libraries(MetallicCPUKernelTests PRIVATE MetallicRuntimeTask)
endif()
if(WIN32)
    target_compile_definitions(MetallicCPUKernelTests PRIVATE NOMINMAX)
endif()
add_test(NAME MetallicCPUKernelTests COMMAND MetallicCPUKernelTests)
set_tests_properties(MetallicCPUKernelTests PROPERTIES LABELS "cpu;simd" TIMEOUT 30)

add_executable(MetallicCPUKernelBench AABBTransformBench.cpp)
target_link_libraries(MetallicCPUKernelBench PRIVATE MetallicCPUKernels)
